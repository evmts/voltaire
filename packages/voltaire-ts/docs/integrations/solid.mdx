`@tevm/voltaire/provider` and `@tevm/voltaire/jsonrpc` integrate cleanly with SolidJS `createResource`, signals, and SolidStart `'use server'` functions.

## Installation

```bash
npm install @tevm/voltaire
```

## Provider Context

```tsx
// src/context/ethereum.tsx
import { HttpProvider } from '@tevm/voltaire/provider';
import { createContext, useContext, type ParentComponent } from 'solid-js';

const EthereumContext = createContext<HttpProvider>();

export const EthereumProvider: ParentComponent<{ url: string }> = (props) => {
  const provider = new HttpProvider(props.url);
  return <EthereumContext.Provider value={provider}>{props.children}</EthereumContext.Provider>;
};

export function useProvider(): HttpProvider {
  const provider = useContext(EthereumContext);
  if (!provider) throw new Error('useProvider must be used inside EthereumProvider');
  return provider;
}
```

## Fetching Data with `createResource`

```tsx
// src/components/Balance.tsx
import { Address } from '@tevm/voltaire/Address';
import { Wei } from '@tevm/voltaire/Denomination';
import { Rpc } from '@tevm/voltaire/jsonrpc';
import { createResource, Show } from 'solid-js';
import { useProvider } from '../context/ethereum';

export function Balance(props: { address: string }) {
  const provider = useProvider();

  const [balance, { refetch }] = createResource(
    () => (Address.isValid(props.address) ? props.address : undefined),
    async (address) => {
      const hex = (await provider.request(
        Rpc.Eth.GetBalanceRequest(Address.from(address), 'latest'),
      )) as string;
      return Wei.toEther(Wei.from(BigInt(hex)));
    },
  );

  return (
    <Show when={!balance.loading} fallback={<p>Loading...</p>}>
      <Show when={!balance.error} fallback={<p>Error: {balance.error.message}</p>}>
        <p>Balance: {balance()} ETH</p>
        <button onClick={refetch}>Refresh</button>
      </Show>
    </Show>
  );
}
```

## SolidStart Server Function

```tsx
// src/routes/address/[address].tsx
import { createAsync, type RouteDefinition } from '@solidjs/router';
import { Address } from '@tevm/voltaire/Address';
import { Wei } from '@tevm/voltaire/Denomination';
import { Rpc } from '@tevm/voltaire/jsonrpc';
import { HttpProvider } from '@tevm/voltaire/provider';
import { Show } from 'solid-js';

async function getAddressData(address: string) {
  'use server';
  const provider = new HttpProvider(process.env.RPC_URL ?? 'https://eth.llamarpc.com');
  const addr = Address.from(address);

  const [balanceHex, nonceHex, code] = (await Promise.all([
    provider.request(Rpc.Eth.GetBalanceRequest(addr, 'latest')),
    provider.request(Rpc.Eth.GetTransactionCountRequest(addr, 'latest')),
    provider.request(Rpc.Eth.GetCodeRequest(addr, 'latest')),
  ])) as [string, string, string];

  return {
    address,
    balance: Wei.toEther(Wei.from(BigInt(balanceHex))),
    nonce: Number(BigInt(nonceHex)),
    isContract: code !== '0x',
  };
}

export const route: RouteDefinition = {
  preload: ({ params }) => getAddressData(params.address),
};

export default function AddressPage(props: { params: { address: string } }) {
  const data = createAsync(() => getAddressData(props.params.address));

  return (
    <Show when={data()}>
      {(d) => (
        <div>
          <h1>{d().address}</h1>
          <p>Balance: {d().balance} ETH</p>
          <p>Nonce: {d().nonce}</p>
          <p>Type: {d().isContract ? 'Contract' : 'EOA'}</p>
        </div>
      )}
    </Show>
  );
}
```

## Block Head Polling

```tsx
import { Rpc } from '@tevm/voltaire/jsonrpc';
import { createSignal, onCleanup } from 'solid-js';
import { useProvider } from '../context/ethereum';

export function useLatestBlock() {
  const provider = useProvider();
  const [blockNumber, setBlockNumber] = createSignal<bigint | null>(null);

  const poll = async () => {
    const hex = (await provider.request(Rpc.Eth.BlockNumberRequest())) as string;
    setBlockNumber(BigInt(hex));
  };

  void poll();
  const timer = setInterval(() => void poll(), 12_000);
  onCleanup(() => clearInterval(timer));

  return blockNumber;
}
```
