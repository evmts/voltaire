`@tevm/voltaire/provider` and `@tevm/voltaire/jsonrpc` run in Astro frontmatter (build-time static generation), server API routes, and client islands without Node-only dependencies.

## Installation

```bash
npm install @tevm/voltaire
```

## Shared Provider Module

```typescript
// src/lib/provider.ts
import { HttpProvider } from '@tevm/voltaire/provider';

export const provider = new HttpProvider(
  import.meta.env.RPC_URL ?? 'https://eth.llamarpc.com',
);
```

## Server-Rendered Page

```astro
---
// src/pages/address/[address].astro
import { Address } from '@tevm/voltaire/Address';
import { Wei } from '@tevm/voltaire/Denomination';
import { Rpc } from '@tevm/voltaire/jsonrpc';
import { provider } from '../../lib/provider';

export const prerender = false;

const { address } = Astro.params;
if (!address || !Address.isValid(address)) {
  return Astro.redirect('/404');
}

const addr = Address.from(address);
const [balanceHex, nonceHex, code] = (await Promise.all([
  provider.request(Rpc.Eth.GetBalanceRequest(addr, 'latest')),
  provider.request(Rpc.Eth.GetTransactionCountRequest(addr, 'latest')),
  provider.request(Rpc.Eth.GetCodeRequest(addr, 'latest')),
])) as [string, string, string];

const data = {
  address,
  balance: Wei.toEther(Wei.from(BigInt(balanceHex))),
  nonce: Number(BigInt(nonceHex)),
  isContract: code !== '0x',
};
---

<html>
  <body>
    <h1>{data.address}</h1>
    <p>Balance: {data.balance} ETH</p>
    <p>Nonce: {data.nonce}</p>
    <p>Type: {data.isContract ? 'Contract' : 'EOA'}</p>
  </body>
</html>
```

## API Route with Cache-Control

```typescript
// src/pages/api/balance/[address].ts
import { Address } from '@tevm/voltaire/Address';
import { Wei } from '@tevm/voltaire/Denomination';
import { Rpc } from '@tevm/voltaire/jsonrpc';
import type { APIRoute } from 'astro';
import { provider } from '../../../lib/provider';

export const prerender = false;

export const GET: APIRoute = async ({ params }) => {
  const { address } = params;
  if (!address || !Address.isValid(address)) {
    return Response.json({ error: 'Invalid Ethereum address' }, { status: 400 });
  }

  try {
    const hex = (await provider.request(
      Rpc.Eth.GetBalanceRequest(Address.from(address), 'latest'),
    )) as string;
    return Response.json(
      { address, balance: Wei.toEther(Wei.from(BigInt(hex))) },
      { headers: { 'Cache-Control': 'public, max-age=12' } },
    );
  } catch (e) {
    return Response.json(
      { error: e instanceof Error ? e.message : 'RPC error' },
      { status: 502 },
    );
  }
};
```

## Static Pre-Rendering (`getStaticPaths`)

```astro
---
// src/pages/contracts/[address].astro
import { Address } from '@tevm/voltaire/Address';
import { Wei } from '@tevm/voltaire/Denomination';
import { Rpc } from '@tevm/voltaire/jsonrpc';
import { provider } from '../../lib/provider';

export function getStaticPaths() {
  return [
    '0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2', // WETH
    '0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48', // USDC
  ].map((address) => ({ params: { address } }));
}

const { address } = Astro.params;
const balanceHex = (await provider.request(
  Rpc.Eth.GetBalanceRequest(Address.from(address!), 'latest'),
)) as string;
---

<p>{address}: {Wei.toEther(Wei.from(BigInt(balanceHex)))} ETH</p>
```
