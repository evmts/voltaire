`@tevm/voltaire/provider` and `@tevm/voltaire/jsonrpc` integrate directly with Vue 3 `ref`/`watch` composables and Nuxt server routes with zero runtime adapters.

## Installation

```bash
npm install @tevm/voltaire
```

## Provider Injection

Provide a single `HttpProvider` at app startup and inject it into composables:

```typescript
// plugins/ethereum.ts
import { HttpProvider } from '@tevm/voltaire/provider';
import { inject, type App, type InjectionKey } from 'vue';

export const providerKey: InjectionKey<HttpProvider> = Symbol('voltaire-provider');

export function installEthereum(app: App, rpcUrl = 'https://eth.llamarpc.com') {
  app.provide(providerKey, new HttpProvider(rpcUrl));
}

export function useProvider(): HttpProvider {
  const provider = inject(providerKey);
  if (!provider) throw new Error('Voltaire HttpProvider not provided');
  return provider;
}
```

## Reactive Balance Composable

```typescript
// composables/useBalance.ts
import { Address } from '@tevm/voltaire/Address';
import { Wei } from '@tevm/voltaire/Denomination';
import { Rpc } from '@tevm/voltaire/jsonrpc';
import { ref, toValue, watchEffect, type MaybeRefOrGetter } from 'vue';
import { useProvider } from '../plugins/ethereum';

export function useBalance(address: MaybeRefOrGetter<string | undefined>) {
  const provider = useProvider();
  const balance = ref<string | null>(null);
  const loading = ref(false);
  const error = ref<Error | null>(null);

  async function refresh() {
    const raw = toValue(address);
    if (!raw || !Address.isValid(raw)) {
      balance.value = null;
      return;
    }

    loading.value = true;
    error.value = null;
    try {
      const hex = (await provider.request(
        Rpc.Eth.GetBalanceRequest(Address.from(raw), 'latest'),
      )) as string;
      balance.value = Wei.toEther(Wei.from(BigInt(hex)));
    } catch (e) {
      error.value = e instanceof Error ? e : new Error(String(e));
    } finally {
      loading.value = false;
    }
  }

  watchEffect(() => {
    void refresh();
  });

  return { balance, loading, error, refresh };
}
```

```vue
<!-- components/BalanceDisplay.vue -->
<script setup lang="ts">
import { useBalance } from '@/composables/useBalance';

const props = defineProps<{ address: string }>();
const { balance, loading, error, refresh } = useBalance(() => props.address);
</script>

<template>
  <div v-if="loading">Loading...</div>
  <div v-else-if="error">Error: {{ error.message }}</div>
  <div v-else>
    <p>Balance: {{ balance }} ETH</p>
    <button @click="refresh">Refresh</button>
  </div>
</template>
```

## Nuxt Server Route

Fetch account state server-side without exposing private RPC API keys to browsers:

```typescript
// server/api/address/[address].ts
import { Address } from '@tevm/voltaire/Address';
import { Wei } from '@tevm/voltaire/Denomination';
import { Rpc } from '@tevm/voltaire/jsonrpc';
import { HttpProvider } from '@tevm/voltaire/provider';

const provider = new HttpProvider(process.env.RPC_URL ?? 'https://eth.llamarpc.com');

export default defineEventHandler(async (event) => {
  const address = getRouterParam(event, 'address');
  if (!address || !Address.isValid(address)) {
    throw createError({ statusCode: 400, message: 'Invalid Ethereum address' });
  }

  const addr = Address.from(address);
  const [balanceHex, nonceHex, code] = (await Promise.all([
    provider.request(Rpc.Eth.GetBalanceRequest(addr, 'latest')),
    provider.request(Rpc.Eth.GetTransactionCountRequest(addr, 'latest')),
    provider.request(Rpc.Eth.GetCodeRequest(addr, 'latest')),
  ])) as [string, string, string];

  return {
    address,
    balance: Wei.toEther(Wei.from(BigInt(balanceHex))),
    nonce: Number(BigInt(nonceHex)),
    isContract: code !== '0x',
  };
});
```
