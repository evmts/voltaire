## Install & Core Imports

```typescript
import { Effect, Layer, Schedule, Stream, Exit } from 'effect'
import * as S from 'effect/Schema'
import * as Address from 'voltaire-effect/primitives/Address'
import {
  getBalance, getBlockNumber, getBlock,
  Provider, HttpTransport,
  withTimeout, withRetrySchedule
} from 'voltaire-effect'
```

## Decode -> Operate -> Provide

```typescript
const ProviderLayer = Provider.pipe(
  Layer.provide(HttpTransport('https://eth.llamarpc.com'))
)

const program = Effect.gen(function* () {
  const addr = yield* S.decode(Address.Hex)(input)
  return yield* getBalance(addr, 'latest')
})

const balance = await Effect.runPromise(program.pipe(Effect.provide(ProviderLayer)))
```

## Schema Operations

| Operation | Call | Return Type |
| --- | --- | --- |
| Decode sync | `S.decodeSync(Address.Hex)('0x...')` | `AddressType` (throws `ParseError`) |
| Decode effect | `S.decode(Address.Hex)('0x...')` | `Effect<AddressType, ParseError>` |
| Decode either | `S.decodeEither(Address.Hex)('0x...')` | `Either<AddressType, ParseError>` |
| Encode sync | `S.encodeSync(Address.Hex)(addr)` | `` `0x${string}` `` |
| Encode checksum | `S.encode(Address.Checksummed)(addr)` | `Effect<string, ParseError, KeccakService>` |

## Error Handling

```typescript
program.pipe(
  Effect.catchTag('ParseError', () => Effect.succeed(fallback)),
  Effect.catchTags({
    TransportError: (e) => Effect.succeed(0n),
    ProviderResponseError: (e) => Effect.fail(e)
  })
)
```

## Provider Free Functions

Provider operations are free functions that read `ProviderService` from context:

```typescript
import {
  getBlockNumber, getBalance, getBlock, getTransaction,
  getTransactionReceipt, getLogs, call, estimateGas
} from 'voltaire-effect'

yield* getBlockNumber()
yield* getBalance(addr, 'latest')
yield* getBlock({ blockTag: 'latest' })
yield* getTransaction(txHash)
yield* getTransactionReceipt(txHash)
yield* getLogs({ address, topics, fromBlock, toBlock })
yield* call({ to, data })
yield* estimateGas({ to, data, value })
```

## Contracts & Signer

```typescript
// Contract factory
const token = yield* Contract(address, erc20Abi)
yield* token.read.balanceOf(user)
yield* token.simulate.transfer(to, amount)
yield* token.write.transfer(to, amount)
yield* token.getEvents('Transfer', { fromBlock, toBlock })

// Signer service
const signer = yield* SignerService
yield* signer.sendTransaction({ to, value: 1_000_000_000_000_000_000n })
yield* signer.signMessage(Hex.fromString('Hello'))
yield* signer.signTypedData(typedData)
```

## Layer Composition

```typescript
// Read-only provider
const ProviderLayer = Provider.pipe(
  Layer.provide(HttpTransport('https://eth.llamarpc.com'))
)

// Full local signer stack
const CryptoLayer = Layer.mergeAll(Secp256k1Live, KeccakLive)
const SignerLayer = Layer.mergeAll(
  Signer.Live,
  CryptoLayer,
  ProviderLayer
).pipe(Layer.provideMerge(LocalAccount(privateKey)))

// Single provide at the entry point
await Effect.runPromise(program.pipe(Effect.provide(SignerLayer)))
```

## Concurrency & Collections

| Pattern | Behavior |
| --- | --- |
| `Effect.all({ a, b })` | Runs named effects, returns typed object |
| `Effect.all([a, b], { concurrency: 5 })` | Runs tuple/array with bounded concurrency |
| `Effect.forEach(items, fn, { concurrency: 5 })` | Maps collection effectfully with bounded concurrency |
| `Effect.forEach(items, fn, { discard: true })` | Runs side effects without allocating result array |

```typescript
const { blockNumber, balance, chainId } = yield* Effect.all({
  blockNumber: getBlockNumber(),
  balance: getBalance(addr),
  chainId: getChainId()
})

const balances = yield* Effect.forEach(addresses, (a) => getBalance(a), {
  concurrency: 5
})
```

## Per-Request Overrides & Transports

```typescript
// FiberRef per-request overrides
getBalance(addr).pipe(
  withTimeout('5 seconds'),
  withRetrySchedule(
    Schedule.exponential('500 millis').pipe(
      Schedule.jittered,
      Schedule.compose(Schedule.recurs(3))
    )
  )
)

// Transport configurations
HttpTransport('https://eth.llamarpc.com')
HttpTransport({ url, timeout: '60 seconds', batch: { batchSize: 50, wait: 10 } })
WebSocketTransport('wss://eth.llamarpc.com')
BrowserTransport
```

## Streaming, Crypto & Debug

```typescript
// Block streaming
import { makeBlockStream } from 'voltaire-effect/services'
const blockStream = yield* makeBlockStream()
yield* Stream.runForEach(
  blockStream.watch({ include: 'transactions' }),
  (event) => Effect.log(`Event: ${event.type}`)
)

// Crypto services
import { CryptoLive, KeccakService, Secp256k1Service } from 'voltaire-effect/crypto'
const keccak = yield* KeccakService
const hash = yield* keccak.hash(data)

// Debug & Engine APIs
const debug = yield* DebugService
const trace = yield* debug.traceTransaction('0x...')
const engine = yield* EngineApiService
const caps = yield* engine.exchangeCapabilities(['engine_newPayloadV3'])
```

## Testing & Runners

```typescript
import { TestTransport, Provider } from 'voltaire-effect'

const TestLayer = Provider.pipe(
  Layer.provide(
    TestTransport({
      eth_blockNumber: () => '0x3039',
      eth_getBalance: () => '0xde0b6b3a7640000'
    })
  )
)

await Effect.runPromise(program.pipe(Effect.provide(TestLayer)))
```

| Runner | Output | Failure Behavior |
| --- | --- | --- |
| `Effect.runPromise(eff)` | `Promise<A>` | Rejects on failure or defect |
| `Effect.runPromiseExit(eff)` | `Promise<Exit<A, E>>` | Captures full `Cause<E>` without throwing |
| `Effect.runSync(eff)` | `A` | Throws on async boundary or failure |
| `Effect.runSyncEither(eff)` | `Either<A, E>` | Returns `Left(e)` on typed failure |
