```bash bun
bun add voltaire-effect @tevm/voltaire effect
```

## Runtime Support

| Runtime | Transports | Crypto & HD Wallet |
| --- | --- | --- |
| Node.js 20+ / Bun 1.1+ | `HttpTransport`, `WebSocketTransport`, `TestTransport` | Full WASM + native FFI (including `MnemonicAccount`) |
| Browser / Edge | `HttpTransport`, `WebSocketTransport`, `BrowserTransport` (EIP-1193) | WASM crypto; native FFI HD wallet unavailable |

Requires TypeScript 5.4+ with `strict: true`.

## 1. Fetch the Latest Block Number

Compose `HttpTransport` into `Provider` once, then provide the composed layer at the entry point:

```typescript
import { Effect, Layer } from 'effect'
import { getBlockNumber, Provider, HttpTransport } from 'voltaire-effect'

const ProviderLayer = Provider.pipe(
  Layer.provide(HttpTransport('https://eth.llamarpc.com'))
)

const program = Effect.gen(function* () {
  return yield* getBlockNumber()
})

const blockNumber = await Effect.runPromise(
  program.pipe(Effect.provide(ProviderLayer))
)
console.log(`Block: ${blockNumber}`)
```

## 2. Decode Branded Primitives

Schemas validate raw input and return Voltaire's branded `Uint8Array` types directly:

```typescript
import * as S from 'effect/Schema'
import * as Address from 'voltaire-effect/primitives/Address'
import * as A from '@tevm/voltaire/Address'

const addr = S.decodeSync(Address.Hex)('0x742d35Cc6634C0532925a3b844Bc9e7595f251e3')

A.equals(addr, addr) // true
A.isZero(addr)       // false
A.toHex(addr)        // "0x742d35cc6634c0532925a3b844bc9e7595f251e3"
```

For fallible decoding without throwing, use `S.decode` and handle `ParseError`:

```typescript
import { Effect } from 'effect'
import * as S from 'effect/Schema'
import * as Address from 'voltaire-effect/primitives/Address'
import * as A from '@tevm/voltaire/Address'

const parseAddress = (input: string) =>
  S.decode(Address.Hex)(input).pipe(
    Effect.catchTag('ParseError', () => Effect.succeed(A.zero()))
  )
```

<Note>
Most provider and contract APIs accept `AddressInput` (either a branded `AddressType` or a `0x${string}` literal). Decode untrusted strings at application boundaries and pass `AddressType` internally.
</Note>

## 3. EIP-55 Checksummed Addresses

EIP-55 checksumming hashes the lowercase hex with Keccak-256, so encoding `Address.Checksummed` declares `KeccakService` in its requirements channel:

```typescript
import { Effect } from 'effect'
import * as S from 'effect/Schema'
import * as Address from 'voltaire-effect/primitives/Address'
import { KeccakLive } from 'voltaire-effect/crypto/Keccak256'

const addr = S.decodeSync(Address.Hex)('0x742d35cc6634c0532925a3b844bc9e7595f251e3')

const checksummed = await Effect.runPromise(
  S.encode(Address.Checksummed)(addr).pipe(Effect.provide(KeccakLive))
)
// "0x742d35Cc6634C0532925a3b844Bc9e7595f251e3"
```

## 4. Read a Smart Contract

Pass a `const` ABI to `Contract()` to infer method names, parameter types, and return types:

```typescript
import { Effect, Layer } from 'effect'
import { Contract, Provider, HttpTransport } from 'voltaire-effect'

const ProviderLayer = Provider.pipe(
  Layer.provide(HttpTransport('https://eth.llamarpc.com'))
)

const erc20Abi = [
  {
    type: 'function',
    name: 'balanceOf',
    stateMutability: 'view',
    inputs: [{ name: 'account', type: 'address' }],
    outputs: [{ type: 'uint256' }]
  }
] as const

const readUsdcBalance = Effect.gen(function* () {
  const usdc = yield* Contract('0xA0b86991c6218b36c1d19D4a2e9Eb0cE3606eB48', erc20Abi)
  return yield* usdc.read.balanceOf('0xd8dA6BF26964aF9D7eEd9e03E53415D37aA96045')
})

const balance = await Effect.runPromise(
  readUsdcBalance.pipe(Effect.provide(ProviderLayer))
)
```

## 5. Per-Request Timeouts and Retries

Configure transport-wide defaults on `HttpTransport`, and override individual calls with FiberRef helpers (`withTimeout`, `withRetrySchedule`):

```typescript
import { Schedule } from 'effect'
import { getBalance, HttpTransport, withTimeout, withRetrySchedule } from 'voltaire-effect'

// Transport-level defaults
const TransportLayer = HttpTransport({
  url: 'https://eth.llamarpc.com',
  timeout: '30 seconds',
  retrySchedule: Schedule.exponential('500 millis').pipe(
    Schedule.jittered,
    Schedule.compose(Schedule.recurs(5))
  )
})

// Scoped per-request override
const fastBalance = getBalance(addr).pipe(
  withTimeout('5 seconds'),
  withRetrySchedule(
    Schedule.exponential('250 millis').pipe(
      Schedule.jittered,
      Schedule.compose(Schedule.recurs(3))
    )
  )
)
```
