This page compares idiomatic usage of Voltaire's Zig 0.15 API against Voltaire TypeScript (`@tevm/voltaire`), Alloy (Rust), and go-ethereum (Go).

## Runtime & Memory Model

| Property | Voltaire (Zig) | Voltaire (TypeScript) | Alloy (Rust) | go-ethereum (Go) |
| :--- | :--- | :--- | :--- | :--- |
| Memory management | Explicit `std.mem.Allocator`; stack arrays for fixed types | GC (`Uint8Array` views) | Ownership / stack `[u8; N]` | GC (`[N]byte` + slices) |
| `Address` representation | `struct { bytes: [20]u8 }` | Branded 20-byte `Uint8Array` | `FixedBytes<20>` | `[20]byte` |
| `Hash` representation | `[32]u8` | Branded 32-byte `Uint8Array` | `FixedBytes<32>` | `[32]byte` |
| Compile-time hex validation | `Address.addressFromHex` / `Hash.fromHexComptime` | Type-level / runtime | `address!()` macro | Runtime only |
| Error model | Error unions (`!T`) | Exceptions | `Result<T, E>` | `(T, error)` |
| WASM target | Native `wasm32-wasi` reactor | Consumes Zig WASM | `wasm32-unknown-unknown` | `js/wasm` (GC runtime) |

## Address Parsing & EIP-55 Checksumming

<Tabs>
  <Tab title="Voltaire (Zig)">
```zig
const primitives = @import("primitives");
const Address = primitives.Address;

// Runtime parsing (validates length and hex digits)
const addr = try Address.fromHex("0x742d35cc6634c0532925a3b844bc9e7595f51e3e");

// Stack-allocated [42]u8 EIP-55 checksum (zero heap allocation)
const checksum = Address.toChecksummed(addr);
// "0x742d35Cc6634C0532925a3b844Bc9e7595f51e3e"

// Compile-time literal parsing (fails build on bad hex)
const static_addr = comptime Address.addressFromHex("0x742d35cc6634c0532925a3b844bc9e7595f51e3e");

// Byte-level equality
const same = Address.equals(addr, static_addr); // true
const valid = Address.isValidChecksum(&checksum); // true
```
  </Tab>
  <Tab title="Voltaire (TypeScript)">
```typescript
import { Address } from '@tevm/voltaire';

const addr = Address('0x742d35cc6634c0532925a3b844bc9e7595f51e3e');
const checksum = Address.toChecksummed(addr);
// "0x742d35Cc6634C0532925a3b844Bc9e7595f51e3e"

const other = Address('0x742d35Cc6634C0532925a3b844Bc9e7595f51e3e');
const same = Address.equals(addr, other); // true
```
  </Tab>
  <Tab title="Alloy (Rust)">
```rust
use alloy_primitives::{address, Address};

let addr: Address = "0x742d35cc6634c0532925a3b844bc9e7595f51e3e".parse()?;
let checksum = addr.to_checksum(None);

let static_addr = address!("742d35cc6634c0532925a3b844bc9e7595f51e3e");
let same = addr == static_addr;
```
  </Tab>
  <Tab title="go-ethereum (Go)">
```go
import "github.com/ethereum/go-ethereum/common"

addr := common.HexToAddress("0x742d35cc6634c0532925a3b844bc9e7595f51e3e")
checksum := addr.Hex() // EIP-55 checksummed
valid := common.IsHexAddress("0x742d35cc6634c0532925a3b844bc9e7595f51e3e")
```
  </Tab>

## Keccak-256 Hashing & Selectors

<Tabs>
  <Tab title="Voltaire (Zig)">
```zig
const primitives = @import("primitives");
const crypto = @import("crypto");

// Stack-only 32-byte digest
const digest: [32]u8 = primitives.Hash.keccak256("hello");
const hex: [66]u8 = crypto.Hash.toHex(digest);
// "0x1c8aff950685c2ed4bc3174f3472287b56d9517b9c948127319a09a7a36deac8"

// 4-byte function selector (stack-only)
const selector: [4]u8 = crypto.Hash.selectorFromSignature("transfer(address,uint256)");
// [4]u8{ 0xa9, 0x05, 0x9c, 0xbb }

// EIP-191 personal message digest (allocates temporary prefix buffer, frees before returning)
const personal = try crypto.Hash.eip191HashMessage("Hello, Ethereum!", allocator);
```
  </Tab>
  <Tab title="Voltaire (TypeScript)">
```typescript
import { Keccak256, Hex } from '@tevm/voltaire';

const digest = Keccak256.hashString('hello');
const hex = Hex.fromBytes(digest);
// "0x1c8aff950685c2ed4bc3174f3472287b56d9517b9c948127319a09a7a36deac8"

const selector = Keccak256.selector('transfer(address,uint256)');
```
  </Tab>
  <Tab title="Alloy (Rust)">
```rust
use alloy_primitives::keccak256;

let digest = keccak256(b"hello");
let selector: [u8; 4] = keccak256(b"transfer(address,uint256)")[..4].try_into().unwrap();
```
  </Tab>
  <Tab title="go-ethereum (Go)">
```go
import "github.com/ethereum/go-ethereum/crypto"

digest := crypto.Keccak256Hash([]byte("hello"))
selector := crypto.Keccak256([]byte("transfer(address,uint256)"))[:4]
```

## RLP Encoding & Decoding

<Tabs>
  <Tab title="Voltaire (Zig)">
```zig
const primitives = @import("primitives");
const Rlp = primitives.Rlp;

const items = [_][]const u8{ "cat", "dog" };
const encoded = try Rlp.encode(allocator, items[0..]);
defer allocator.free(encoded);
// 0xc88363617483646f67

// stream = false rejects trailing bytes
const decoded = try Rlp.decode(allocator, encoded, false);
defer decoded.data.deinit(allocator);

switch (decoded.data) {
    .List => |list| {
        const first = list[0].String; // "cat"
        _ = first;
    },
    .String => unreachable,
}
```
  </Tab>
  <Tab title="Voltaire (TypeScript)">
```typescript
import { Rlp } from '@tevm/voltaire';

const enc = new TextEncoder();
const encoded = Rlp.encode([enc.encode('cat'), enc.encode('dog')]);
const decoded = Rlp.decode(encoded);
```
  </Tab>
  <Tab title="Alloy (Rust)">
```rust
use alloy_rlp::{Decodable, Encodable};

let items = vec![b"cat".as_slice(), b"dog".as_slice()];
let mut buf = Vec::new();
items.encode(&mut buf);
```
  </Tab>
  <Tab title="go-ethereum (Go)">
```go
import "github.com/ethereum/go-ethereum/rlp"

encoded, err := rlp.EncodeToBytes([]string{"cat", "dog"})
var decoded []string
err = rlp.DecodeBytes(encoded, &decoded)
```
  </Tab>

## ABI Function Calldata Encoding

<Tabs>
  <Tab title="Voltaire (Zig)">
```zig
const primitives = @import("primitives");
const abi = primitives.AbiEncoding;

const selector = abi.computeSelector("transfer(address,uint256)");
const recipient = try primitives.Address.fromHex("0x742d35Cc6634C0532925a3b844Bc9e7595f51e3e");
const params = [_]abi.AbiValue{
    abi.addressValue(recipient),
    abi.uint256_value(1_000_000_000_000_000_000),
};

const calldata = try abi.encodeFunctionData(allocator, selector, &params);
defer allocator.free(calldata);

const types = [_]abi.AbiType{ .address, .uint256 };
const decoded = try abi.decodeFunctionData(allocator, calldata, &types);
defer allocator.free(decoded.parameters);
```
  </Tab>
  <Tab title="Voltaire (TypeScript)">
```typescript
import { Abi } from '@tevm/voltaire';

const abi = Abi([
  {
    type: 'function',
    name: 'transfer',
    stateMutability: 'nonpayable',
    inputs: [
      { name: 'to', type: 'address' },
      { name: 'amount', type: 'uint256' },
    ],
    outputs: [{ name: '', type: 'bool' }],
  },
] as const);

const calldata = abi.encode('transfer', [
  '0x742d35Cc6634C0532925a3b844Bc9e7595f51e3e',
  1_000_000_000_000_000_000n,
]);
```
